📋 SQL setup script 
drop table if exists audit_log cascade;
drop table if exists cases cascade;
create table cases (
  id uuid default gen_random_uuid() primary key,
  num text not null, name text not null,
  id_number text, date_of_birth date,
  tribe text, village text, address text,
  type text, description text, assistance text,
  status text default 'Open', contacts text,
  created_at timestamptz default now(),
  updated_at timestamptz default now(),
  created_by uuid references auth.users(id),
  created_by_name text,
  updated_by uuid references auth.users(id),
  updated_by_name text
);
create table audit_log (
  id uuid default gen_random_uuid() primary key,
  case_id uuid, case_num text, action text not null,
  performed_by uuid references auth.users(id),
  performed_by_name text,
  performed_at timestamptz default now(),
  details jsonb
);
alter table cases enable row level security;
alter table audit_log enable row level security;
create policy "Auth users full access on cases" on cases for all to authenticated using (true) with check (true);
create policy "Auth users full access on audit_log" on audit_log for all to authenticated using (true) with check (true);
